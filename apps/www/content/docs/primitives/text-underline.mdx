## Usage

```tsx
import { TextUnderline } from '@/components/sora-ui/texts/text-underline';

export default function Page() {
  return (
    <TextUnderline href="#" underlineColor="brand">
      Shop Performance Collection
    </TextUnderline>
  );
}
```
