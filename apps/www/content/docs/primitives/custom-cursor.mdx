## Usage

```tsx
import {
  CustomCursor,
  CustomCursorTarget,
} from '@/components/sora-ui/effects/custom-cursor';

export default function Page() {
  return (
    <CustomCursor color="#ff4c24">
      <CustomCursorTarget>
        <YourLogo />
      </CustomCursorTarget>
    </CustomCursor>
  );
}
```
