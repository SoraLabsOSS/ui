## Usage

```tsx
import { DrawUnderlineLink } from '@/components/sora-ui/texts/draw-underline-link';

export default function Page() {
  return (
    <DrawUnderlineLink href="#" underlineColor="#ff4c24">
      Hover me
    </DrawUnderlineLink>
  );
}
```
