## Usage

```tsx
import { DualWipeReveal } from '@/components/sora-ui/texts/dual-wipe-reveal';

export function Example() {
  return (
    <h2 className="text-4xl font-semibold tracking-tight">
      <DualWipeReveal
        accentColor="bg-emerald-500"
        wipeColor="bg-neutral-800"
      >
        Find it, copy it, ship it.
      </DualWipeReveal>
    </h2>
  );
}
```

Stagger multiple lines:

```tsx
<div className="flex flex-col gap-2">
  <DualWipeReveal delay={0}>Find it,</DualWipeReveal>
  <DualWipeReveal delay={0.15}>copy it,</DualWipeReveal>
  <DualWipeReveal delay={0.3}>ship it.</DualWipeReveal>
</div>
```

Reverse wipe direction (right to left):

```tsx
<DualWipeReveal direction="right">
  Revealed from right to left
</DualWipeReveal>
```

Vertical wipe reveals:

```tsx
<DualWipeReveal direction="down">
  Swept from top to bottom
</DualWipeReveal>
```
