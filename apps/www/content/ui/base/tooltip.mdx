## Installation

<ComponentInstallation name="base-tooltip">
  <Step>
    <h4 className="pt-1 pb-4">Add the TooltipProvider to the root of your app.</h4>

    ```tsx title="app/layout.tsx"
    import { TooltipProvider } from "@/components/sora-ui/base/tooltip";

    export default function RootLayout({
      children,
    }: {
      children: React.ReactNode;
    }) {
      return (
        <html lang="en">
          <body>
            <TooltipProvider>{children}</TooltipProvider>
          </body>
        </html>
      );
    }
    ```
  </Step>
</ComponentInstallation>

## Usage

```tsx
import { Button } from "@/components/sora-ui/base/button";
import {
  Tooltip,
  TooltipArrow,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@/components/sora-ui/base/tooltip";
import { BookmarkIcon } from "lucide-react";

export default function Page() {
  return (
    <TooltipProvider delay={100}>
      <Tooltip>
        <TooltipTrigger
          render={
            <Button size="icon" variant="outline">
              <BookmarkIcon className="size-4" />
              <span className="sr-only">Add to library</span>
            </Button>
          }
        />
        <TooltipContent side="top">
          Add to library
          <TooltipArrow />
        </TooltipContent>
      </Tooltip>
    </TooltipProvider>
  );
}
```
